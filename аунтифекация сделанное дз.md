using Microsoft.AspNetCore.Authentication;
using Microsoft.AspNetCore.Authentication.Cookies;
using Microsoft.AspNetCore.Authentication.JwtBearer;
using Microsoft.IdentityModel.Tokens;
using System.IdentityModel.Tokens.Jwt;
using System.Security.Claims;
using System.Text;

var builder = WebApplication.CreateBuilder(args);

var jwtKey = builder.Configuration["Jwt:Key"] ?? "SUPER_SECRET_KEY_AT_LEAST_32_BYTES_LONG_1234";
var jwtIssuer = builder.Configuration["Jwt:Issuer"] ?? "https://myapp.example";
var jwtAudience = builder.Configuration["Jwt:Audience"] ?? "myapp.api";

builder.Services.AddDataProtection()
    .PersistKeysToFileSystem(new DirectoryInfo("/var/keys"))
    .SetApplicationName("MyApp");

builder.Services
    .AddAuthentication(options =>
    {
        options.DefaultScheme = "Smart";
        options.DefaultChallengeScheme = "Smart";
    })
    .AddPolicyScheme("Smart", "Smart", options =>
    {
        options.ForwardDefaultSelector = ctx =>
            ctx.Request.Headers.Authorization.ToString()
                .StartsWith("Bearer ", StringComparison.OrdinalIgnoreCase)
                    ? "Bearer"
                    : "Cookies";
    })
    .AddCookie("Cookies", options =>
    {
        options.Cookie.Name = ".MyApp.Auth";
        options.Cookie.HttpOnly = true;
        options.Cookie.SecurePolicy = CookieSecurePolicy.Always;
        options.Cookie.SameSite = SameSiteMode.Lax;
        options.ExpireTimeSpan = TimeSpan.FromHours(8);
        options.SlidingExpiration = true;
        options.LoginPath = "/login";
        options.LogoutPath = "/logout";
        options.AccessDeniedPath = "/forbidden";
        options.Events = new CookieAuthenticationEvents
        {
            OnValidatePrincipal = async ctx =>
            {
                var sessionId = ctx.Principal?.FindFirst("sid")?.Value;
                if (sessionId is null || RevokedSessions.IsRevoked(sessionId))
                {
                    ctx.RejectPrincipal();
                    await ctx.HttpContext.SignOutAsync("Cookies");
                }
            }
        };
    })
    .AddJwtBearer("Bearer", options =>
    {
        options.RequireHttpsMetadata = true;
        options.SaveToken = false;
        options.TokenValidationParameters = new TokenValidationParameters
        {
            ValidateIssuer = true,
            ValidIssuer = jwtIssuer,
            ValidateAudience = true,
            ValidAudience = jwtAudience,
            ValidateIssuerSigningKey = true,
            IssuerSigningKey = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(jwtKey)),
            ValidateLifetime = true,
            ClockSkew = TimeSpan.FromSeconds(30),
            NameClaimType = JwtRegisteredClaimNames.Sub,
            RoleClaimType = "role"
        };
        options.Events = new JwtBearerEvents
        {
            OnTokenValidated = ctx =>
            {
                var jti = ctx.Principal?.FindFirst(JwtRegisteredClaimNames.Jti)?.Value;
                if (jti is not null && RevokedTokens.IsRevoked(jti))
                    ctx.Fail("Token revoked");
                return Task.CompletedTask;
            }
        };
    });

builder.Services.AddAuthorization(options =>
{
    options.AddPolicy("Html", p => p
        .AddAuthenticationSchemes("Cookies")
        .RequireAuthenticatedUser());
    options.AddPolicy("Api", p => p
        .AddAuthenticationSchemes("Bearer")
        .RequireAuthenticatedUser());
});

builder.Services.AddAntiforgery();

record User(string Username, string PasswordHash, string[] Roles);

static class UserStore
{
    public static readonly Dictionary<string, User> Users = new()
    {
        ["alice"] = new User("alice", Hash("p@ss"), new[] { "user", "admin" }),
        ["bob"] = new User("bob", Hash("p@ss"), new[] { "user" })
    };

    public static string Hash(string s) =>
        Convert.ToBase64String(System.Security.Cryptography.SHA256.HashData(
            System.Text.Encoding.UTF8.GetBytes(s)));

    public static bool Verify(string u, string p) =>
        Users.TryGetValue(u, out var user) && user.PasswordHash == Hash(p);
}

static class TokenService
{
    public static string CreateAccessToken(
        string username,
        IEnumerable<string> roles,
        string key,
        string issuer,
        string audience,
        TimeSpan lifetime)
    {
        var claims = new List<Claim>
        {
            new(JwtRegisteredClaimNames.Sub, username),
            new(JwtRegisteredClaimNames.Jti, Guid.NewGuid().ToString("N")),
            new(JwtRegisteredClaimNames.Iat,
                DateTimeOffset.UtcNow.ToUnixTimeSeconds().ToString(),
                ClaimValueTypes.Integer64)
        };
        claims.AddRange(roles.Select(r => new Claim("role", r)));

        var securityKey = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(key));
        var creds = new SigningCredentials(securityKey, SecurityAlgorithms.HmacSha256);

        var token = new JwtSecurityToken(
            issuer: issuer,
            audience: audience,
            claims: claims,
            notBefore: DateTime.UtcNow,
            expires: DateTime.UtcNow.Add(lifetime),
            signingCredentials: creds);

        return new JwtSecurityTokenHandler().WriteToken(token);
    }

    public static string CreateRefreshToken()
    {
        var bytes = new byte[64];
        System.Security.Cryptography.RandomNumberGenerator.Fill(bytes);
        return Convert.ToBase64String(bytes);
    }
}

static class RefreshStore
{
    public record Entry(string Username, DateTimeOffset Expires);
    private static readonly Dictionary<string, Entry> _map = new();

    public static void Save(string user, string token, DateTimeOffset exp) =>
        _map[token] = new Entry(user, exp);

    public static Entry? Get(string token) =>
        _map.TryGetValue(token, out var e) ? e : null;

    public static void Revoke(string token) => _map.Remove(token);
}

static class RevokedTokens
{
    private static readonly HashSet<string> _set = new();
    public static bool IsRevoked(string jti) => _set.Contains(jti);
    public static void Revoke(string jti) => _set.Add(jti);
}

static class RevokedSessions
{
    private static readonly HashSet<string> _set = new();
    public static bool IsRevoked(string sid) => _set.Contains(sid);
    public static void Revoke(string sid) => _set.Add(sid);
}

var app = builder.Build();

app.UseHttpsRedirection();
app.UseAuthentication();
app.UseAuthorization();
app.UseAntiforgery();

app.MapGet("/login", (HttpContext ctx) =>
{
    var tokens = ctx.RequestServices.GetRequiredService<Microsoft.AspNetCore.Antiforgery.IAntiforgery>();
    var pair = tokens.GetAndStoreTokens(ctx);
    return Results.Ok(new { csrf = pair.RequestToken });
});

app.MapPost("/login", async (HttpContext ctx, LoginDto dto) =>
{
    if (!UserStore.Verify(dto.Username, dto.Password))
        return Results.Unauthorized();

    var user = UserStore.Users[dto.Username];
    var sid = Guid.NewGuid().ToString("N");

    var claims = new List<Claim>
    {
        new(ClaimTypes.Name, user.Username),
        new(ClaimTypes.NameIdentifier, user.Username),
        new("sid", sid)
    };
    claims.AddRange(user.Roles.Select(r => new Claim(ClaimTypes.Role, r)));

    var identity = new ClaimsIdentity(claims, "Cookies");
    var principal = new ClaimsPrincipal(identity);

    await ctx.SignInAsync("Cookies", principal, new AuthenticationProperties
    {
        IsPersistent = dto.RememberMe,
        ExpiresUtc = DateTimeOffset.UtcNow.AddHours(dto.RememberMe ? 24 * 7 : 8),
        AllowRefresh = true
    });

    return Results.Ok(new { user = user.Username, sid });
}).RequireAntiforgery();

app.MapPost("/logout", async (HttpContext ctx, [FromForm] string? sid) =>
{
    if (sid is not null)
        RevokedSessions.Revoke(sid);
    await ctx.SignOutAsync("Cookies");
    return Results.Ok();
}).RequireAntiforgery();

app.MapPost("/token", (TokenRequest req) =>
{
    if (!UserStore.Verify(req.Username, req.Password))
        return Results.Unauthorized();

    var user = UserStore.Users[req.Username];
    var access = TokenService.CreateAccessToken(
        user.Username, user.Roles, jwtKey, jwtIssuer, jwtAudience,
        TimeSpan.FromMinutes(15));

    var refresh = TokenService.CreateRefreshToken();
    RefreshStore.Save(user.Username, refresh, DateTimeOffset.UtcNow.AddDays(7));

    return Results.Ok(new
    {
        access_token = access,
        refresh_token = refresh,
        token_type = "Bearer",
        expires_in = 900
    });
});

app.MapPost("/token/refresh", (RefreshRequest req) =>
{
    var entry = RefreshStore.Get(req.RefreshToken);
    if (entry is null || entry.Expires < DateTimeOffset.UtcNow)
        return Results.Unauthorized();

    RefreshStore.Revoke(req.RefreshToken);

    var user = UserStore.Users[entry.Username];
    var access = TokenService.CreateAccessToken(
        user.Username, user.Roles, jwtKey, jwtIssuer, jwtAudience,
        TimeSpan.FromMinutes(15));
    var newRefresh = TokenService.CreateRefreshToken();
    RefreshStore.Save(user.Username, newRefresh, DateTimeOffset.UtcNow.AddDays(7));

    return Results.Ok(new
    {
        access_token = access,
        refresh_token = newRefresh,
        expires_in = 900
    });
});

app.MapGet("/me", (HttpContext ctx) =>
    ctx.User.Identity?.IsAuthenticated == true
        ? Results.Ok(new
        {
            name = ctx.User.Identity!.Name,
            sub = ctx.User.FindFirst(JwtRegisteredClaimNames.Sub)?.Value,
            roles = ctx.User.FindAll(ClaimTypes.Role)
                .Concat(ctx.User.FindAll("role"))
                .Select(c => c.Value)
                .Distinct()
        })
        : Results.Unauthorized());

app.MapGet("/admin", () => Results.Ok("admin area"))
    .RequireAuthorization(p => p.RequireRole("admin").RequireClaim("role", "admin"));

app.MapGet("/dashboard", () => "html")
    .RequireAuthorization("Html");

app.MapGet("/api/data", () => new { ok = true })
    .RequireAuthorization("Api");

app.Run();
